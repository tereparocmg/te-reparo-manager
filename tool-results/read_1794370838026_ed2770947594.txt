    95→    95→  const [modalHistorial, setModalHistorial] = useState(false)

    98→    98→
    99→    99→  const cargar = useCallback(async () => {
   100→   100→    if (!tallerId) return
   101→   101→    setCargando(true)
   102→   102→    try {
   103→   103→      const params = new URLSearchParams({ tallerId })
   104→   104→      if (filtros.q) params.set('q', filtros.q)
   105→   105→      if (filtros.categoria) params.set('categoria', filtros.categoria)
   106→   106→      if (filtros.tipo) params.set('tipo', filtros.tipo)
   107→   107→      if (filtros.stockBajo) params.set('stockBajo', filtros.stockBajo)
   108→   108→      if (filtros.activo) params.set('activo', filtros.activo)
   109→   109→      const res = await api.get<{ productos: Producto[]; categorias: string[] }>(`/api/productos?${params}`)
   110→   110→      setProductos(res.productos)
   111→   111→      setCategorias(res.categorias)
   112→   112→    } catch (e) {
   113→   113→      toast.error(e instanceof Error ? e.message : 'Error al cargar inventario')
   114→   114→    } finally {
   115→   115→      setCargando(false)
   116→   116→    }
   117→   117→  }, [tallerId, filtros])
   118→   118→
   119→   119→  useEffect(() => {
   120→   120→    cargar()
   121→   121→  }, [cargar])
   122→   122→
   123→   123→  useEffect(() => {
   124→   124→    if (esAdmin) {
   125→   125→      api.get<{ talleres: Taller[] }>('/api/talleres').then((r) => setListaTalleres(r.talleres)).catch(() => {})
   126→   126→    }
   127→   127→  }, [esAdmin])

   156→   156→        costo: String(p.costo ?? 0),
   157→   157→        stockMinimo: String(p.stockMinimo ?? 0),
   158→   158→        pagoOperarioTipo: p.pagoOperarioTipo ?? 'FIJO',
   159→   159→        pagoOperarioValor: String(p.pagoOperarioValor ?? 0),
   160→   160→        garantiaDias: String(p.garantiaDias ?? 0),
   161→   161→      },
   162→   162→    })

   164→   164→  }

   166→   166→  async function guardarProducto() {
   167→   167→    const d = productoEdit.datos
   168→   168→    if (d.nombre.trim().length < 2 || !d.categoria.trim() || !d.precio) {
   169→   169→      toast.error('Complete nombre, categoría y precio')
   170→   170→      return
   171→   171→    }
   172→   172→    try {
   173→   173→      const cuerpo: Record<string, unknown> = {
   174→   174→        nombre: d.nombre.trim(),
   175→   175→        categoria: d.categoria.trim(),
   176→   176→        tipo: d.tipo,
   177→   177→        precio: parseFloat(d.precio),
   178→   178→        monedaPrecio: d.monedaPrecio,
   179→   179→        unidad: d.unidad,
   180→   180→      }
   181→   181→      if (esAdmin) {
   182→   182→        cuerpo.costo = parseFloat(d.costo || '0')
   183→   183→        cuerpo.stockMinimo = parseInt(d.stockMinimo || '0', 10)
   184→   184→        cuerpo.pagoOperarioTipo = d.pagoOperarioTipo
   185→   185→        cuerpo.pagoOperarioValor = parseFloat(d.pagoOperarioValor || '0')
   186→   186→        cuerpo.garantiaDias = parseInt(d.garantiaDias || '0', 10)
   187→   187→      }
   188→   188→      if (!productoEdit.id) {
   189→   189→        await api.post(`/api/productos`, { ...cuerpo, tallerId, cantidad: esAdmin ? parseInt(d.cantidadInicial || '0', 10) : 0 })
   190→   190→        toast.success('Producto creado')
   191→   191→      } else {
   192→   192→        await api.patch(`/api/productos?id=${productoEdit.id}`, cuerpo)
   193→   193→        toast.success('Producto actualizado')
   194→   194→      }
   195→   195→      setModalProducto(false)
   196→   196→      cargar()
   197→   197→    } catch (e) {
   198→   198→      toast.error(e instanceof Error ? e.message : 'Error al guardar producto')
   199→   199→    }
   200→   200→  }
   201→   201→
   202→   202→  async function toggleActivo(p: Producto) {

   208→   208→      toast.error(e instanceof Error ? e.message : 'Error')

   210→   210→  }

   228→   228→    }
   229→   229→  }

   253→   253→        productoId: productoAccion.id,
   254→   254→        tallerDestinoId: formTransfer.tallerDestinoId,

   259→   259→      setModalTransfer(false)

   263→   263→      toast.error(e instanceof Error ? e.message : 'Error al transferir')
   264→   264→    }
   265→   265→  }
   266→   266→
   267→   267→  const productosVisibles = useMemo(() => productos, [productos])
   268→   268→
   269→   269→  // ---- Ctrl+S guarda el formulario/modal abierto ----
   270→   270→  useGuardarConCtrlS(
   271→   271→    () => {
   272→   272→      if (modalProducto) guardarProducto()
   273→   273→      else if (modalEntrada) registrarEntrada()
   274→   274→      else if (modalBaja) registrarBaja()
   275→   275→      else if (modalTransfer) registrarTransferencia()
   276→   276→    },
   277→   277→    modalProducto || modalEntrada || modalBaja || modalTransfer
   278→   278→  )
   279→   279→
   280→   280→  // ---- Navegación por teclado en la tabla (↑↓ Enter Supr) ----
   281→   281→  const suprArmado = useRef<{ i: number; t: number } | null>(null)
   282→   282→  function suprFila(i: number) {
   283→   283→    if (!esAdmin) return
   284→   284→    const p = productosVisibles[i]
   285→   285→    if (!p) return
   286→   286→    if (!p.activo) {
   287→   287→      toast.info('El producto ya está inactivo')
   288→   288→      return
   289→   289→    }
   290→   290→    const ahora = Date.now()
   291→   291→    if (suprArmado.current?.i === i && ahora - suprArmado.current.t < 4000) {
   292→   292→      suprArmado.current = null
   293→   293→      toggleActivo(p)
   294→   294→    } else {
   295→   295→      suprArmado.current = { i, t: ahora }
   296→   296→      toast.warning(`Pulse Supr de nuevo para desactivar «${p.nombre}»`, { duration: 4000 })
   297→   297→    }
   298→   298→  }

   300→   300→    total: productosVisibles.length,
   301→   301→    onAbrir: (i) => abrirEditar(productosVisibles[i]),
   302→   302→    onEliminar: suprFila,
   303→   303→  })
   304→   304→
   305→   305→  return (
   306→   306→    <div className="space-y-4">
   307→   307→      <div className="flex items-center justify-between flex-wrap gap-2">

   309→   309→        <Button onClick={abrirNuevo}>
   310→   310→          <Plus className="h-4 w-4 mr-1" /> Nuevo producto
   311→   311→        </Button>
   312→   312→      </div>
   313→   313→
   314→   314→      {/* Filtros */}
   315→   315→      <Card>
   316→   316→        <CardContent className="p-3 flex flex-wrap gap-2 items-center">

   321→   321→              onChange={(e) => setFiltros((f) => ({ ...f, q: e.target.value }))}
   322→   322→              placeholder="Buscar por nombre o código…"
   323→   323→              className="pl-8"
   324→   324→              aria-label="Buscar producto"
   325→   325→            />
   326→   326→          </div>
   327→   327→          <Select value={filtros.categoria} onValueChange={(v) => setFiltros((f) => ({ ...f, categoria: v === '__todos__' ? '' : v }))}>
   328→   328→            <SelectTrigger className="w-[160px]" aria-label="Categoría">
   329→   329→              <SelectValue placeholder="Categoría" />
   330→   330→            </SelectTrigger>
   331→   331→            <SelectContent>
   332→   332→              <SelectItem value="__todos__">Todas</SelectItem>
   333→   333→              {categorias.map((c) => (
   334→   334→                <SelectItem key={c} value={c}>{c}</SelectItem>
   335→   335→              ))}
   336→   336→            </SelectContent>
   337→   337→          </Select>
   338→   338→          <Select value={filtros.tipo} onValueChange={(v) => setFiltros((f) => ({ ...f, tipo: v === '__todos__' ? '' : v }))}>
   339→   339→            <SelectTrigger className="w-[170px]" aria-label="Tipo">
   340→   340→              <SelectValue placeholder="Tipo" />
   341→   341→            </SelectTrigger>
   342→   342→            <SelectContent>
   343→   343→              <SelectItem value="__todos__">Todos</SelectItem>
   344→   344→              {TIPOS.map((t) => (
   345→   345→                <SelectItem key={t} value={t}>{TIPO_PRODUCTO_LABEL[t]}</SelectItem>

   347→   347→            </SelectContent>
   348→   348→          </Select>

   350→   350→            <SelectTrigger className="w-[150px]" aria-label="Stock bajo">
   351→   351→              <SelectValue placeholder="Stock" />
   352→   352→            </SelectTrigger>
   353→   353→            <SelectContent>
   354→   354→              <SelectItem value="__todos__">Todo el stock</SelectItem>
   355→   355→              <SelectItem value="true">Solo stock bajo</SelectItem>
   356→   356→            </SelectContent>
   357→   357→          </Select>
   358→   358→          <Select value={filtros.activo} onValueChange={(v) => setFiltros((f) => ({ ...f, activo: v === '__todos__' ? '' : v }))}>
   359→   359→            <SelectTrigger className="w-[140px]" aria-label="Estado">
   360→   360→              <SelectValue placeholder="Estado" />
   361→   361→            </SelectTrigger>
   362→   362→            <SelectContent>
   363→   363→              <SelectItem value="true">Activos</SelectItem>
   364→   364→              <SelectItem value="false">Inactivos</SelectItem>
   365→   365→              <SelectItem value="__todos__">Todos</SelectItem>
   366→   366→            </SelectContent>
   367→   367→          </Select>
   368→   368→        </CardContent>
   369→   369→      </Card>
   370→   370→
   371→   371→      {/* Tabla */}
   372→   372→      <Card>
   373→   373→        <CardContent className="p-0 overflow-x-auto">
   374→   374→          <Table>
   375→   375→            <TableHeader>
   376→   376→              <TableRow>
   377→   377→                <TableHead>Producto</TableHead>
   378→   378→                <TableHead>Tipo</TableHead>
   379→   379→                <TableHead className="text-right">Stock</TableHead>
   380→   380→                <TableHead className="text-right">Precio</TableHead>
   381→   381→                {esAdmin && <TableHead className="text-right">Costo</TableHead>}
   382→   382→                {esAdmin && <TableHead className="text-right">Pago operario</TableHead>}
   383→   383→                <TableHead>Estado</TableHead>
   384→   384→                <TableHead className="text-right">Acciones</TableHead>
   385→   385→              </TableRow>
   386→   386→            </TableHeader>
   387→   387→            <TableBody data-tabla="filas">
   388→   388→              {productosVisibles.map((p, i) => (
   389→   389→                <TableRow
   390→   390→                  key={p.id}
   391→   391→                  className={`${!p.activo ? 'opacity-60' : ''} ${i === filaSel ? 'ring-2 ring-ring/60 ring-inset' : ''}`}
   392→   392→                  {...propsFila(i)}
   393→   393→                >
   394→   394→                  <TableCell>
   395→   395→                    <div className="font-medium">{p.nombre}</div>
   396→   396→                    <div className="text-xs text-muted-foreground font-mono">
   397→   397→                      {p.codigoInterno} · {p.categoria}
   398→   398→                    </div>
   399→   399→                  </TableCell>
   400→   400→                  <TableCell>
   401→   401→                    <Badge variant={esServicio(p.tipo) ? 'outline' : 'secondary'} className="text-xs">
   402→   402→                      {esServicio(p.tipo) ? 'Servicio' : 'Producto'}
   403→   403→                    </Badge>
   404→   404→                  </TableCell>
   405→   405→                  <TableCell className="text-right num-tabular">

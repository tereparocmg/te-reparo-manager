    83→/** ¿El foco está en un campo de texto? (los atajos de líneas no deben interferir) */
    84→function focoEnCampo(): boolean {
    85→  const el = document.activeElement
    86→  if (!el) return false
    87→  const tag = el.tagName
    88→  return tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT' || (el as HTMLElement).getAttribute('role') === 'option'
    89→}

   328→  /** Validaciones previas a confirmar la venta (devuelve mensaje de error o null). */
   329→  function validarVenta(): string | null {
   330→    if (!carrito || !tallerId) return 'No hay carrito o taller'
   331→    if (carrito.items.length === 0) return 'El carrito está vacío'
   332→    if (itemsInvalidos.length > 0) return 'Hay líneas inválidas: servicios sin operario o cantidad mayor que el stock'
   333→    const pagosValidos = pagosActivos.filter((p) => parseFloat(p.monto) > 0)
   334→    if (pagosValidos.length === 0) return 'Registre al menos un pago'
   335→    for (const p of pagosValidos) {
   336→      if (p.metodo === 'ZELLE' && (p.moneda !== 'USD' || !p.referencia.trim())) {
   337→        return 'Zelle solo en USD y con referencia obligatoria'
   338→      }
   339→    }
   340→    if (cubierto + 0.001 < totales.cup) return `Faltan ${fmtCUP(faltante)} por cubrir`
   341→    return null
   342→  }

   344→  function abrirCobro() {
   345→    if (!carrito || carrito.items.length === 0) {
   346→      toast.error('El carrito está vacío')
   347→      return
   348→    }
   349→    // Si no hay pagos, precarga uno por el monto exacto
   350→    if ((pagos[carrito.id] ?? []).length === 0) agregarPago()
   351→    setModalCobro(true)
   352→  }

   354→  async function confirmarVenta(e?: React.FormEvent) {
   355→    e?.preventDefault()
   356→    const error = validarVenta()
   357→    if (error) {
   358→      toast.error(error)
   359→      sonidoError()
   360→      return
   361→    }
   362→    if (!carrito || !tallerId) return
   363→    setConfirmando(true)
   364→    try {
   365→      const pagosValidos = pagosActivos.filter((p) => parseFloat(p.monto) > 0)
   366→      const res = await api.post<{ venta: { id: string; codigo: string } }>('/api/ventas', {
   367→        tallerId,
   368→        clienteId: carrito.clienteId,
   369→        items: carrito.items.map((it) => ({
   370→          productoId: it.productoId,
   371→          cantidad: it.cantidad,
   372→          precioUnitario: it.precio !== it.precioOriginal ? it.precio : undefined,
   373→          operarioId: it.operarioId,
   374→        })),
   375→        pagos: pagosValidos.map((p) => ({
   376→          metodo: p.metodo,
   377→          moneda: p.moneda,
   378→          monto: parseFloat(p.monto),
   379→          referencia: p.metodo === 'ZELLE' ? p.referencia.trim() : p.referencia || undefined,
   380→        })),
   381→      })
   382→      const detalle = await api.get<VentaDetalle>(`/api/ventas?id=${res.venta.id}`)
   383→      setTicketData(detalle)
   384→      setModalCobro(false)
   385→      toast.success(`Venta ${res.venta.codigo} registrada`)
   386→      sonidoVenta()
   387→      cerrarCarrito(carrito.id)
   388→      setPagos((p) => {
   389→        const copia = { ...p }
   390→        delete copia[carrito.id]
   391→        return copia
   392→      })
   393→      nuevoCarrito(tallerId)
   394→      setLineaSel(-1)
   395→      await cargarProductos()
   396→      // El buscador vuelve a quedar listo para el siguiente escaneo
   397→      setTimeout(() => busquedaRef.current?.focus(), 50)
   398→    } catch (err) {
   399→      toast.error(err instanceof Error ? err.message : 'Error al registrar la venta')
   400→      sonidoError()
   401→    } finally {
   402→      setConfirmando(false)
   403→    }
   404→  }

   420→  const cambiarCantidadSel = useCallback(
   421→    (delta: number) => {
   422→      if (!carrito || lineaSel < 0) return

   424→      if (!it) return
   425→      if (esServicio(it.tipo)) {
   426→        toast.info('Los servicios se cobran ×1')
   427→        return
   428→      }
   429→      const nueva = it.cantidad + delta
   430→      if (nueva < 1) return
   431→      if (nueva > it.stock) {
   432→        toast.error(`Stock máximo: ${it.stock}`)
   433→        return
   434→      }
   435→      cambiarCantidad(carrito.id, it.productoId, nueva)
   436→    },
